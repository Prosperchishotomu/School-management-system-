15. UI/UX & Routing Specification (React 18 + React Router 6)
15.1 Design Principles
Depth without getting lost — every URL mirrors the information hierarchy; the UI always shows where you are (breadcrumb + active nav glow) and how to go up (one‑click parent segments).
Calm futurism — dark‑capable glass surfaces, subtle gradients and glow on interaction only; no decorative animation that slows data work.
Data first — grids, marks entry, and ledgers are the heroes: dense, keyboard‑driven, offline‑tolerant.
Progressive disclosure — list → detail → action; heavy tools (imports, publishing, reconciliation) open in focused workspaces with steppers.
Zero‑reload navigation — SPA route transitions with shared layouts; state persists while moving between sibling tabs.

Mechanics

Guards: <RequireRoles roles> and <RequireFeature flag> wrappers on layout routes; failure → <Navigate to="/403" state={{from}}/> / <ModuleDisabledPanel/> rendered inside the parent outlet.
Breadcrumbs: from useMatches() → each match's meta.title (strings or functions of params) → <Breadcrumbs/>; data loaders (loader) resolve human labels (student name, invoice no) so crumbs never show UUIDs.
Deep links & sharing: every filter/sort/page/tab is URL state via useSearchParams; copying the URL reproduces the exact view. Query params validated with Zod → invalid values silently fall back to defaults.
Code splitting: one lazy chunk per portal branch + per heavyweight page (timetable editor, marks entry, reconciliation); prefetch on sidebar hover (IntersectionObserver/router.prefetch).
Transitions: <AnimatePresence mode="popLayout"> around <Outlet/>; grid row → detail uses layoutId shared‑element expansion.
Scroll restoration per route; ScrollRestoration component from data router.
Offline (PWA): route loaders read from IndexedDB cache when navigator.onLine === false (attendance/marks/results snapshots); amber "Offline — showing cached data" bar with Sync now action; queued writes badge in topbar.
Errors: route errorElement boundary per portal (not global) → friendly panel + retry + "go to dashboard"; API MODULE_DISABLED maps to the module panel.

src/routes/guards.tsx

tsx

import { Navigate, useLocation, Outlet } from 'react-router-dom'
import { useSession } from '#/store/session'
import { ModuleDisabledPanel } from '#/components/data/ModuleDisabledPanel'
import type { Role } from './types'

export function RequireRoles({ roles }: { roles?: Role[] }) {
  const { user } = useSession()
  const location = useLocation()
  if (!user) return <Navigate to="/login" state={{ from: location.pathname + location.search }} replace />
  if (roles && !roles.some(r => user.roles.includes(r)))
    return <Navigate to="/403" state={{ from: location.pathname }} replace />
  return <Outlet />
}

export function RequireFeature({ feature }: { feature: string }) {
  const flags = useSession(s => s.flags)          // from GET /auth/me
  if (!flags?.[feature]) return <ModuleDisabledPanel feature={feature} />  // renders in-place (§15.5)
  return <Outlet />
}

import { useSession } from '#/store/session'

const BASE = import.meta.env.VITE_API_URL ?? '/api/v1'

export class ApiError extends Error {
  constructor(public status: number, public code: string, message: string,
              public fields?: Record<string, string>, public requestId?: string) { super(message) }
}

let refreshing: Promise<boolean> | null = null

async function refresh(): Promise<boolean> {
  refreshing ??= (async () => {
    try {
      const res = await fetch(`${BASE}/auth/refresh`, { method: 'POST', credentials: 'include' })
      if (!res.ok) return false
      const { data } = await res.json()
      useSession.getState().setAccessToken(data.access_token)
      return true
    } finally { refreshing = null }
  })()
  return refreshing
}

export interface RequestOptions {
  method?: 'GET'|'POST'|'PATCH'|'PUT'|'DELETE'
  body?: unknown
  query?: Record<string, unknown>
  idempotencyKey?: string          // required by server for payments/invoice-gen/imports
  signal?: AbortSignal
  retried?: boolean
}

export async function request<T = any>(path: string, opts: RequestOptions = {}): Promise<T> {
  const { method = 'GET', body, query, idempotencyKey, signal, retried } = opts
  const url = new URL(BASE + path, window.location.origin)
  for (const [k, v] of Object.entries(query ?? {}))
    if (v !== undefined && v !== '' && v !== null) url.searchParams.set(k, String(v))

  const token = useSession.getState().accessToken
  const requestId = crypto.randomUUID()

  const res = await fetch(url, {
    method, signal, credentials: 'include',
    headers: {
      ...(body ? { 'Content-Type': 'application/json' } : {}),
      ...(token ? { Authorization: `Bearer ${token}` } : {}),
      'X-Request-ID': requestId,
      ...(idempotencyKey ? { 'Idempotency-Key': idempotencyKey } : {}),
    },
    body: body ? JSON.stringify(body) : undefined,
  })

  // 401 → single-flight refresh → retry once → else hard logout
  if (res.status === 401 && !retried) {
    if (await refresh()) return request(path, { ...opts, retried: true })
    useSession.getState().logout()
    throw new ApiError(401, 'TOKEN_EXPIRED', 'Session expired', undefined, requestId)
  }

  const json = await res.json().catch(() => ({}))
  if (!res.ok) {
    const e = json.error ?? {}
    throw new ApiError(res.status, e.code ?? 'INTERNAL_ERROR', e.message ?? res.statusText,
                       e.fields, json.request_id ?? requestId)
  }
  return (json.data !== undefined || json.meta !== undefined) ? json : json
}

import { ApiError, api } from './api'

/* Minimal IndexedDB object store: queue of pending write ops */
const DB_NAME = 'sms-offline', STORE = 'queue'

function open(): Promise<IDBDatabase> {
  return new Promise((res, rej) => {
    const r = indexedDB.open(DB_NAME, 1)
    r.onupgradeneeded = () => { r.result.createObjectStore(STORE, { keyPath: 'op_id' }) }
    r.onsuccess = () => res(r.result)
    r.onerror = () => rej(r.error)
  })
}
async function store(mode: IDBTransactionMode) {
  const db = await open()
  return db.transaction(STORE, mode).objectStore(STORE)
}
const asPromise = <T,>(r: IDBRequest<T>) => new Promise<T>((res, rej) => { r.onsuccess = () => res(r.result); r.onerror = () => rej(r.error) })

export interface QueueOp {
  op_id: string                                  // client uuid → idempotency on replay (§5, §10.6)
  path: string
  body: unknown
  meta: Record<string, unknown>                  // e.g. { student_id, component_id } for cell sync
  enqueued_at: string
}

export async function enqueueOp(op: Omit<QueueOp, 'enqueued_at'>) {
  ;(await store('readwrite')).put({ ...op, enqueued_at: new Date().toISOString() })
  broadcastCount()
}
async function broadcastCount() {
  const n = await count()
  window.dispatchEvent(new CustomEvent('sms:queue', { detail: { count: n } }))
}
export const count = async () => asPromise(await (await store('readonly')).count())
export const list = async () => asPromise<any[]>(await (await store('readonly')).getAll())

/** Drain the queue. Network errors stop the drain; 4xx = permanent reject. */
export async function replayAll(): Promise<void> {
  const ops = await list()
  for (const op of ops) {
    try {
      await api.post(op.path, op.body, { idempotencyKey: op.op_id })
      ;(await store('readwrite')).delete(op.op_id)
      window.dispatchEvent(new CustomEvent('sms:op-saved', { detail: { ...op.meta, ok: true } }))
    } catch (e) {
      if (e instanceof ApiError) {                 // server rejected permanently
        ;(await store('readwrite')).delete(op.op_id)
        window.dispatchEvent(new CustomEvent('sms:op-saved', { detail: { ...op.meta, ok: false, error: e.message } }))
        continue
      }
      break                                        // offline — keep remaining ops queued
    }
  }
  broadcastCount()
}

/* auto-replay on reconnect; banner can also trigger manually */
if (typeof window !== 'undefined') {
  window.addEventListener('online', () => { replayAll().catch(() => {}) })
  broadcastCount()
}

import { routeTree } from './routeTree'
import { useSession } from '#/store/session'
import type { AppRoute, Role } from './types'

export interface NavItem { to: string; label: string; icon?: string; group: string; end: boolean }

export function flattenNav(portal: string, roles: Role[], flags: Record<string, boolean>): NavItem[] {
  const out: NavItem[] = []
  const walk = (nodes: AppRoute[], base: string) => {
    for (const n of nodes) {
      const path = `${base}/${n.path}`.replace(/\/+$/, '') || '/'
      const meta = n.handle?.meta
      if (meta?.roles && !meta.roles.some(r => roles.includes(r))) continue
      if (meta?.feature && !flags[meta.feature]) continue
      if (n.children) {
        if (meta?.nav) out.push({ to: path, label: String(meta.title), icon: meta.icon, group: meta.group ?? 'General', end: false })
        walk(n.children, path)
      } else if (meta?.nav !== false) {
        out.push({ to: path, label: typeof meta?.title === 'function' ? 'Details' : String(meta?.title ?? path),
                   icon: meta?.icon, group: meta?.group ?? 'General', end: true })
      }
    }
  }
  walk(routeTree.find(r => r.path === portal)?.children ?? [], `/${portal}`)
  return out
}

export function requestId(req: Request, res: Response, next: NextFunction) {
  req.id = (req.header('x-request-id') as string) || randomUUID()
  res.setHeader('X-Request-ID', req.id)
  req.log = logger.child({ request_id: req.id })
  const done = res.on('finish', () =>

  res.on('close', () => res.off('finish', done))
  next()
}

// middleware/auth.ts — JWT RS256 + account state (§5, §9.3–9.5)
import jwt from 'jsonwebtoken'
import type { NextFunction, Request, Response } from 'express'
import { AppError } from '../lib/errors'
import { knex } from '../db/knex'
import type { AuthContext } from './requestId'

const permCache = new Map<number, { roles: string[]; permissions: string[]; at: number }>()
const TTL = 60_000

export const requireAuth = async (req: Request, _res: Response, next: NextFunction) => {
  try {
    const token = (req.header('authorization') ?? '').replace(/^Bearer /i, '')
    if (!token) throw new AppError('TOKEN_EXPIRED', 'Missing access token')

    const payload = jwt.verify(token, env.JWT_ACCESS_PUBLIC_KEY, { algorithms: ['RS256'] }) as
      { sub: string; roles: string[]; sid: string; jti: string }

    let perms = permCache.get(Number(payload.sub))
    if (!perms || Date.now() - perms.at > TTL) {
      const roles = await knex('roles')
        .join('user_role', 'user_role.role_id', 'roles.id')
        .where('user_role.user_id', payload.sub).select('roles.slug')
      const permissions = await knex('permissions')
        .join('role_permission', 'role_permission.permission_id', 'permissions.id')
        .join('user_role', 'user_role.role_id', 'role_permission.role_id')
        .where('user_role.user_id', payload.sub).select('permissions.slug')
      perms = { roles: roles.map(r => r.slug), permissions: permissions.map(p => p.slug), at: Date.now() }
      permCache.set(Number(payload.sub), perms)
    }

    const user = await knex('users').where({ id: payload.sub, status: 'active' }).first('id', 'uuid', 'locked_until')
    if (!user) throw new AppError('FORBIDDEN', 'Account is not active')
    if (user.locked_until && new Date(user.locked_until) > new Date()) throw new AppError('ACCOUNT_LOCKED', 'Try again later')

    req.auth = { userId: Number(user.id), uuid: user.uuid, roles: perms.roles, permissions: perms.permissions,
                 sid: payload.sid, jti: payload.jti }
    next()
  } catch (e) {
    next(e instanceof jwt.TokenExpiredError ? new AppError('TOKEN_EXPIRED', 'Session expired')
       : e instanceof jwt.JsonWebTokenError ? new AppError('INVALID_CREDENTIALS', 'Invalid token')
       : e)
  }
}

// middleware/errorHandler.ts — envelope per §5; MySQL driver mapping
import type { NextFunction, Request, Response } from 'express'
import { AppError } from '../lib/errors'

const MYSQL_MAP: Record<string, AppError> = {
  ER_DUP_ENTRY: new AppError('DUPLICATE_RESOURCE', 'A record with these details already exists'),
  ER_NO_REFERENCED_ROW_2: new AppError('VALIDATION_ERROR', 'Referenced record does not exist'),
  ER_ROW_IS_REFERENCED_2: new AppError('BUSINESS_RULE_VIOLATION', 'Cannot delete: dependent records exist'),
}

export function errorHandler(err: unknown, req: Request, res: Response, _next: NextFunction) {
  let e: AppError
  if (err instanceof AppError) e = err
  else {
    const anyErr = err as { code?: string }
    e = (anyErr?.code && MYSQL_MAP[anyErr.code]) ||
        new AppError('INTERNAL_ERROR', env.NODE_ENV === 'production' ? 'Unexpected error' : String((err as Error)?.message))
  }
  if (e.code === 'INTERNAL_ERROR') req.log.error({ err }, 'unhandled')
  else if (e.status >= 400) req.log.warn({ code: e.code }, e.message)

  res.status(e.status).json({
    error: { code: e.code, message: e.message, ...(e.fields ? { fields: e.fields } : {}),
             ...(e.details ? { details: e.details } : {}) },
    request_id: req.id,
  })
}

export function attachSockets(server: http.Server) {
  io = new Server(server, { cors: { origin: env.APP_URL, credentials: true } })

  io.use(async (socket, next) => {
    try {
      const token = socket.handshake.auth?.token as string
      const payload = jwt.verify(token, env.JWT_ACCESS_PUBLIC_KEY, { algorithms: ['RS256'] }) as { sub: string }
      const user = await knex('users').where({ id: payload.sub, status: 'active' }).first('id', 'uuid')
      if (!user) return next(new Error('unauthorized'))
      socket.data.userId = user.id
      socket.data.uuid = user.uuid
      next()
    } catch { next(new Error('unauthorized')) }
  })

  io.on('connection', async socket => {
    await socket.join(`user:${socket.data.uuid}`)
    const roles = (await knex('roles')
      .join('user_role', 'user_role.role_id', 'roles.id')
      .where('user_role.user_id', socket.data.userId).select('roles.slug')).map(r => r.slug)
    await Promise.all(roles.map(r => socket.join(`role:${r}`)))
    socket.on('disconnect', () => logger.debug({ uuid: socket.data.uuid }, 'socket out'))
  })

  return io
}

export const emitToUser = (uuid: string, event: string, payload: unknown) => io?.to(`user:${uuid}`).emit(event, payload)
export const emitToRole = (role: string, event: string, payload: unknown) => io?.to(`role:${role}`).emit(event, payload)
